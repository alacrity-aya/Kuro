## Task 2: Fix Pod Card Position Reset on Refresh

**Files:**
- Modify: `frontend/src/components/topology/TopologyCanvas.tsx`
- Test: Manual browser test

**Problem Analysis:**

In `TopologyCanvas.tsx`, the layout is recalculated every time `topologyNodes` changes:

```tsx
const { nodes: layoutedNodes, edges: layoutedEdges } = useMemo(() => {
  const needsLayout = topologyNodes.some((n) => n.x === undefined || n.y === undefined);
  if (needsLayout) {
    return getLayoutedElements(initialNodes, initialEdges);
  }
  return { nodes: initialNodes, edges: initialEdges };
}, [initialNodes, initialEdges, topologyNodes]);
```

The issue: when nodes are refreshed from API, they may not have x/y coordinates, triggering layout recalculation.

**Step 1: Preserve node positions across refreshes**

Add a ref to store positions:

```tsx
const nodePositionsRef = useRef<Map<string, { x: number; y: number }>>(new Map());
```

**Step 2: Update positions when nodes are dragged**

Add `onNodeDragStop` handler to save positions:

```tsx
const handleNodeDragStop = useCallback(
  (_event: React.MouseEvent, node: Node<CustomNodeData>) => {
    nodePositionsRef.current.set(node.id, node.position);
  },
  []
);
```

**Step 3: Apply saved positions when nodes refresh**

Modify the layout logic:

```tsx
const { nodes: layoutedNodes, edges: layoutedEdges } = useMemo(() => {
  // First, check if we have saved positions for all nodes
  const hasAllPositions = topologyNodes.every(
    (n) => nodePositionsRef.current.has(n.id)
  );
  
  if (hasAllPositions) {
    // Use saved positions
    return {
      nodes: initialNodes.map((n) => ({
        ...n,
        position: nodePositionsRef.current.get(n.id) || n.position,
      })),
      edges: initialEdges,
    };
  }
  
  // Check if nodes have positions from API
  const needsLayout = topologyNodes.some((n) => n.x === undefined || n.y === undefined);
  if (needsLayout) {
    const layouted = getLayoutedElements(initialNodes, initialEdges);
    // Save initial positions
    layouted.nodes.forEach((n) => {
      nodePositionsRef.current.set(n.id, n.position);
    });
    return layouted;
  }
  return { nodes: initialNodes, edges: initialEdges };
}, [initialNodes, initialEdges, topologyNodes]);
```


**Step 1: Create validation utility**

```typescript
// frontend/src/utils/policyValidator.ts

export interface PolicyValidationResult {
  isValid: boolean;
  errors: Record<string, string>;
}

// Bandwidth: number + unit (bps, Kbps, Mbps, Gbps)
const BANDWIDTH_REGEX = /^(\d+(?:\.\d+)?)\s*(bps|Kbps|Mbps|Gbps)$/i;

// Latency/Jitter: number + unit (ms, s)
const LATENCY_REGEX = /^(\d+(?:\.\d+)?)\s*(ms|s)$/i;

// Packet Loss: percentage (0-100%)
const PACKET_LOSS_REGEX = /^(\d+(?:\.\d+)?)\s*%$/;

export function validateBandwidth(value: string): string | null {
  if (!value.trim()) return 'Bandwidth is required';
  if (!BANDWIDTH_REGEX.test(value)) {
    return 'Invalid format. Use: 10Mbps, 1Gbps, etc.';
  }
  return null;
}

export function validateLatency(value: string): string | null {
  if (!value.trim()) return 'Latency is required';
  if (!LATENCY_REGEX.test(value)) {
    return 'Invalid format. Use: 10ms, 1s, etc.';
  }
  return null;
}

export function validateJitter(value: string): string | null {
  if (!value.trim()) return 'Jitter is required';
  if (!LATENCY_REGEX.test(value)) {
    return 'Invalid format. Use: 5ms, 0.5s, etc.';
  }
  return null;
}

export function validatePacketLoss(value: string): string | null {
  if (!value.trim()) return 'Packet loss is required';
  const match = value.match(PACKET_LOSS_REGEX);
  if (!match) {
    return 'Invalid format. Use: 0.1%, 1%, etc.';
  }
  const num = parseFloat(match[1]);
  if (num < 0 || num > 100) {
    return 'Packet loss must be between 0% and 100%';
  }
  return null;
}

export function validatePolicy(policy: {
  bandwidth: string;
  latency: string;
  jitter: string;
  packetLoss: string;
}): PolicyValidationResult {
  const errors: Record<string, string> = {};
  
  const bandwidthError = validateBandwidth(policy.bandwidth);
  if (bandwidthError) errors.bandwidth = bandwidthError;
  
  const latencyError = validateLatency(policy.latency);
  if (latencyError) errors.latency = latencyError;
  
  const jitterError = validateJitter(policy.jitter);
  if (jitterError) errors.jitter = jitterError;
  
  const packetLossError = validatePacketLoss(policy.packetLoss);
  if (packetLossError) errors.packetLoss = packetLossError;
  
  return {
    isValid: Object.keys(errors).length === 0,
    errors,
  };
}
```

## Task 5: Fix TrafficControl Deletion Cleanup

- Modify: `internal/agent/grpc_impl.go`

**Problem Analysis:**

The `traffic_controller.go` handles deletion incorrectly:

```go
if !tc.DeletionTimestamp.IsZero() {
    return ctrl.Result{}, nil  // BUG: Just returns, no cleanup!
}
```

When a TrafficControl is deleted, the BPF rules remain in the kernel map.

**Step 1: Add Finalizer for cleanup**

```go
const (
    trafficControlFinalizer = "simulation.kuro.io/cleanup"
)

func (r *TrafficControlReconciler) Reconcile(ctx context.Context, req ctrl.Request) (ctrl.Result, error) {
    logger := log.FromContext(ctx)

    var tc kurov1alpha1.TrafficControl
    if err := r.Get(ctx, req.NamespacedName, &tc); err != nil {
        return ctrl.Result{}, client.IgnoreNotFound(err)
    }

    // Handle deletion
    if !tc.DeletionTimestamp.IsZero() {
        // Check if finalizer is present
        if controllerutil.ContainsFinalizer(&tc, trafficControlFinalizer) {
            // Cleanup BPF rules
            if err := r.cleanupTrafficControl(ctx, &tc); err != nil {
                logger.Error(err, "Failed to cleanup TrafficControl")
                return ctrl.Result{}, err
            }
            
            // Remove finalizer
            controllerutil.RemoveFinalizer(&tc, trafficControlFinalizer)
            if err := r.Update(ctx, &tc); err != nil {
                return ctrl.Result{}, err
            }
        }
        return ctrl.Result{}, nil
    }

    // Add finalizer if not present
    if !controllerutil.ContainsFinalizer(&tc, trafficControlFinalizer) {
        controllerutil.AddFinalizer(&tc, trafficControlFinalizer)
        if err := r.Update(ctx, &tc); err != nil {
            return ctrl.Result{}, err
        }
    }

    // ... rest of reconcile logic
}
```

**Step 2: Implement cleanup function**

```go
func (r *TrafficControlReconciler) cleanupTrafficControl(ctx context.Context, tc *kurov1alpha1.TrafficControl) error {
    logger := log.FromContext(ctx)
    
    // Find Source Pods
    srcSelector, _ := metav1.LabelSelectorAsSelector(&tc.Spec.Source)
    var srcPods corev1.PodList
    if err := r.List(ctx, &srcPods, client.InNamespace(tc.Namespace), client.MatchingLabelsSelector{Selector: srcSelector}); err != nil {
        return err
    }

    // Find Destination Pods
    dstSelector, _ := metav1.LabelSelectorAsSelector(&tc.Spec.Destination)
    var dstPods corev1.PodList
    if err := r.List(ctx, &dstPods, client.InNamespace(tc.Namespace), client.MatchingLabelsSelector{Selector: dstSelector}); err != nil {
        return err
    }

    // Send cleanup command to each agent
    for _, src := range srcPods.Items {
        if src.Status.PodIP == "" || src.Spec.NodeName == "" {
            continue
        }

        for _, dst := range dstPods.Items {
            if dst.Status.PodIP == "" {
                continue
            }

            // Send DELETE command to agent
            cleanupCmd := domain.LinkPolicy{
                SrcIP: src.Status.PodIP,
                DstIP: dst.Status.PodIP,
                Action: "delete", // New field to indicate deletion
            }
            
            _, err := r.AgentManager.SendCommand(src.Spec.NodeName, tc.Name, cleanupCmd)
            if err != nil {
                logger.Info("Failed to send cleanup command", "node", src.Spec.NodeName, "err", err)
                // Continue trying other nodes
            }
        }
    }

    logger.Info("Cleaned up TrafficControl", "name", tc.Name)
    return nil
}
```
